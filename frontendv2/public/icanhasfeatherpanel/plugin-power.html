Frontend Plugin Power SDK

window.FeatherPanel v2 — events, cancelable actions, modals, search, shortcuts. Works at runtime with no panel pnpm build per plugin.

Quick start (Frontend/index.js)

(function () {
  function boot() {
    var FP = window.FeatherPanel;
    if (!FP || !FP.events) return setTimeout(boot, 50);

    FP.events.on('fp:route:change', function (p) {
      console.log('route', p);
    });

    FP.actions.register('fp:server:power', {
      id: 'myplugin.guard-kill',
      priority: 100,
      handler: async function (ctx, next) {
        if (ctx.action === 'kill' && !confirm('Kill?')) {
          ctx.cancel('aborted');
          return;
        }
        await next();
      },
    });

    FP.search.contribute({
      id: 'myplugin',
      search: function (q) {
        return q ? [{ id: '1', title: 'My plugin', href: '/dashboard' }] : [];
      },
    });

    FP.shortcuts.register({
      id: 'myplugin-toast',
      combo: 'ctrl+shift+m',
      handler: function () { FP.toast.info('Hello from plugin'); },
    });
  }
  boot();
})();

fp:route:changePathname changes
fp:page:mount / fp:page:unmountRoute settle
fp:theme:changeTheme / accent / pack
fp:uipack:changeActive UI pack
fp:session:changeLogin / user change
fp:server:contextEnter/leave server routes

fp:server:console:readyConsole page mounted
fp:files:savedFile save success

Opt-in buttons: <button data-fp-action="fp:my:action" data-fp-payload='{"x":1}'>
